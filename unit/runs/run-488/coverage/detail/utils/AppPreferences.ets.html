export type AppPreferenceValue = string | number; 
 
export interface LoginStateRecord { 
  accountId: string; 
  providerId: string; 
  displayName: string; 
  avatarUri: string; 
  loginTime: number; 
  isLogged: boolean; 
} 
 
export interface AppPreferencesStore { 
  get(key: string, defaultValue: AppPreferenceValue): Promise<AppPreferenceValue>; 
  put(key: string, value: AppPreferenceValue): Promise<void>; 
  flush(): Promise<void>; 
} 
 
export type AppPreferencesLoader = (context: Context, storeName: string) => Promise<AppPreferencesStore>;

const STORE_NAME = 'app_preferences'; 
 
const DEFAULT_APP_PREFERENCES_LOADER: AppPreferencesLoader = async ( 
  context: Context, 
  storeName: string 
): Promise<AppPreferencesStore> => { 
  const store = await dataPreferences.getPreferences(context, storeName); 
  return store as AppPreferencesStore; 
}; 
 
/** 
 * 应用偏好设置工具（EL1） 
 * 
 * 使用前须在 EntryAbility.onCreate 中调用 AppPreferences.init(context) 
 * 
 * ```typescript 
 * await AppPreferences.init(context); 
 * await AppPreferences.set(PrefKey.TMDB_API_KEY, 'your_key'); 
 * const key = await AppPreferences.get(PrefKey.TMDB_API_KEY, ''); 
 * ``` 
 */ 
export class AppPreferences { 
  private static store: AppPreferencesStore | null = null; 
  private static initPromise: Promise<void> | null = null; 
  private static storeLoader: AppPreferencesLoader = DEFAULT_APP_PREFERENCES_LOADER; 
 
  private static generateDeviceUuid(): string { 
    const segment = (): string => { 
      return Math.floor((1 + Math.random()) * 0x10000).toString(16).substring(1); 
    }; 
    return `${segment()}${segment()}-${segment()}-${segment()}-${segment()}-${segment()}${segment()}${segment()}`; 
  } 
 
  static setStoreLoaderForTesting(loader: AppPreferencesLoader | null): void { 
    AppPreferences.storeLoader = loader ?? DEFAULT_APP_PREFERENCES_LOADER; 
    AppPreferences.store = null; 
    AppPreferences.initPromise = null; 
  } 
 
  static resetForTesting(): void { 
    AppPreferences.store = null; 
    AppPreferences.initPromise = null; 
  } 
 
  /** 在 EntryAbility.onCreate 里调用一次 */ 
  static async init(context: Context): Promise<void> { 
    if (AppPreferences.store !== null) { 
      return; 
    } 
    if (AppPreferences.initPromise !== null) { 
      return AppPreferences.initPromise; 
    } 
    AppPreferences.initPromise = (async () => { 
      try { 
        AppPreferences.store = await AppPreferences.storeLoader(context, STORE_NAME); 
        console.info('[AppPreferences] 初始化成功'); 
      } catch (e) { 
        const err = e as BusinessError; 
        console.error(`[AppPreferences] 初始化失败: ${err.message ?? JSON.stringify(e)}`); 
        throw new Error(`AppPreferences init failed: ${err.message}`); 
      } finally { 
        AppPreferences.initPromise = null; 
      } 
    })(); 
    return AppPreferences.initPromise; 
  } 
 
  private static async waitForStore(): Promise<AppPreferencesStore | null> { 
    if (AppPreferences.store !== null) { 
      return AppPreferences.store; 
    } 
    if (AppPreferences.initPromise !== null) { 
      try { 
        await AppPreferences.initPromise; 
      } catch { 
        return null; 
      } 
      return AppPreferences.store; 
    } 
    return null; 
  } 
 
  /** 读取字符串值 */ 
  static async get(key: string, defaultValue: string): Promise<string> { 
    const store = await AppPreferences.waitForStore(); 
    if (!store) { 
      console.warn(`[AppPreferences] 未初始化，key=${key} 返回默认值`); 
      return defaultValue; 
    } 
    try { 
      const val = await store.get(key, defaultValue); 
      return val as string; 
    } catch { 
      return defaultValue; 
    } 
  } 
 
  /** 读取数字值 */ 
  static async getNumber(key: string, defaultValue: number): Promise<number> { 
    const store = await AppPreferences.waitForStore(); 
    if (!store) { 
      console.warn(`[AppPreferences] 未初始化，key=${key} 返回默认值`); 
      return defaultValue; 
    } 
    try { 
      const val = await store.get(key, defaultValue); 
      return val as number; 
    } catch { 
      return defaultValue; 
    } 
  } 
 
  /** 写入值（string 或 number） */ 
  static async set(key: string, value: string | number): Promise<void> { 
    const store = await AppPreferences.waitForStore(); 
    if (!store) { 
      console.warn(`[AppPreferences] 未初始化，key=${key} 写入失败`); 
      return; 
    } 
    try { 
      await store.put(key, value); 
      await store.flush(); 
      console.info(`[AppPreferences] 写入 key=${key}`); 
    } catch (e) { 
      const err = e as BusinessError; 
      console.error(`[AppPreferences] 写入失败 key=${key}: ${err.message ?? JSON.stringify(e)}`); 
    } 
  } 
 
  /** 
   * 获取或创建 OpenSubtitles 代理通道设备 ID。 
   * 首次调用时自动生成 UUID 并持久化；后续调用直接返回已存值。 
   */ 
  static async getOrCreateDeviceId(): Promise<string> { 
    const existing = await AppPreferences.get(PrefKey.OPENSUBTITLES_DEVICE_ID, ''); 
    if (existing.length > 0) { 
      return existing; 
    } 
    const newId = AppPreferences.generateDeviceUuid(); 
    await AppPreferences.set(PrefKey.OPENSUBTITLES_DEVICE_ID, newId); 
    const verified = await AppPreferences.get(PrefKey.OPENSUBTITLES_DEVICE_ID, ''); 
    return verified.length > 0 ? verified : newId; 
  } 
 
  static async getOrCreateDeviceUuid(): Promise<string> { 
    const existingUuid = await AppPreferences.get(PrefKey.METRICS_DEVICE_UUID, ''); 
    if (existingUuid.length > 0) { 
      return existingUuid; 
    } 
    const nextUuid = AppPreferences.generateDeviceUuid(); 
    await AppPreferences.set(PrefKey.METRICS_DEVICE_UUID, nextUuid); 
    const verified = await AppPreferences.get(PrefKey.METRICS_DEVICE_UUID, ''); 
    return verified.length > 0 ? verified : ''; 
  } 
 
  /** 
   * 写入完整登录态（登录成功后调用）。 
   * @param accountId 本 App 内账号 ID 
   * @param providerId provider 标识（'huawei' | 'wechat' ...） 
   * @param displayName 显示昵称（可能为空字符串） 
   * @param avatarUri 头像 URI（可能为空字符串） 
   * @param loginTime 登录时间（毫秒时间戳） 
   */ 
  static async setLoginState( 
    accountId: string, 
    providerId: string, 
    displayName: string, 
    avatarUri: string, 
    loginTime: number 
  ): Promise<void> { 
    const state: LoginStateRecord = { 
      accountId, 
      providerId, 
      displayName, 
      avatarUri, 
      loginTime, 
      isLogged: true 
    }; 
    await AppPreferences.writeLoginState(state); 
  } 
 
  /** 清除登录态（退出登录后调用）；Cloud DB 记录保留。 */ 
  static async clearLoginState(): Promise<void> { 
    const state: LoginStateRecord = { 
      accountId: '', 
      providerId: '', 
      displayName: '', 
      avatarUri: '', 
      loginTime: 0, 
      isLogged: false 
    }; 
    await AppPreferences.writeLoginState(state); 
  } 
 
  private static async writeLoginState(state: LoginStateRecord): Promise<void> { 
    const store = await AppPreferences.waitForStore(); 
    if (store === null) { 
      throw new Error('AppPreferences 未初始化，无法持久化登录态'); 
    } 
    await store.put(PrefKey.ACCOUNT_STATE_RECORD, JSON.stringify(state)); 
    await store.flush(); 
  } 
 
  static async getLoginState(): Promise<LoginStateRecord | null> { 
    const serialized = await AppPreferences.get(PrefKey.ACCOUNT_STATE_RECORD, ''); 
    if (serialized.length === 0) { 
      return null; 
    } 
    try { 
      return JSON.parse(serialized) as LoginStateRecord; 
    } catch (e) { 
      return null; 
    } 
  } 
 
  /** 是否已登录 */ 
  static async isLogged(): Promise<boolean> { 
    const state = await AppPreferences.getLoginState(); 
    if (state !== null) { 
      return state.isLogged; 
    } 
    const legacy = await AppPreferences.getNumber(PrefKey.ACCOUNT_IS_LOGGED, 0); 
    return legacy === 1; 
  } 
 
  /** 当前登录 provider 标识（未登录返回空字符串） */ 
  static async getCurrentProviderId(): Promise<string> { 
    const state = await AppPreferences.getLoginState(); 
    return state?.providerId ?? AppPreferences.get(PrefKey.ACCOUNT_CURRENT_PROVIDER_ID, ''); 
  } 
 
  /** 当前登录账号 ID（未登录返回空字符串） */ 
  static async getAccountId(): Promise<string> { 
    const state = await AppPreferences.getLoginState(); 
    return state?.accountId ?? AppPreferences.get(PrefKey.ACCOUNT_ACCOUNT_ID, ''); 
  } 
 
  /** 当前登录显示昵称（未登录返回空字符串） */ 
  static async getDisplayName(): Promise<string> { 
    const state = await AppPreferences.getLoginState(); 
    return state?.displayName ?? AppPreferences.get(PrefKey.ACCOUNT_DISPLAY_NAME, ''); 
  } 
 
  /** 当前登录头像 URI（未登录返回空字符串） */ 
  static async getAvatarUri(): Promise<string> { 
    const state = await AppPreferences.getLoginState(); 
    return state?.avatarUri ?? AppPreferences.get(PrefKey.ACCOUNT_AVATAR_URI, ''); 
  } 
}
